    ```haskell
    sendEmail :: EmailConfig -> Text -> Text -> Text -> IO (Either Text ())
    sendEmail cfg recipient subject body = do
      let mail = Mail (Address (Just $ ecFromName cfg) (ecFromAddr cfg))
                      [Address Nothing recipient]
                      [] [] subject Nothing [PartText "text/plain" body]
      result <- try $ sendMail (T.unpack $ ecSmtpHost cfg) (ecSmtpPort cfg)
                                (T.unpack $ ecSmtpUser cfg) (T.unpack $ ecSmtpPass cfg) mail
      case result of
        Right _ -> return $ Right ()
        Left e   -> return $ Left (T.pack $ show (e :: SomeException))
    ```
